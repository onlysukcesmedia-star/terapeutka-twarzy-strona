/* ===================== Punkty wejścia ===================== */

function doGet(e) {
  var p = (e && e.parameter) || {};
  try {
    if (p.action === 'availability') {
      return json_(availability_(p.service, p.from, parseInt(p.days || '14', 10)));
    }
    if (p.action === 'services') {
      return json_({ ok: true, services: CONFIG.SERVICES });
    }
    return json_({ ok: true, info: 'Terapeutka Twarzy – API rezerwacji działa.' });
  } catch (err) {
    return json_({ ok: false, error: String(err && err.message || err) });
  }
}

function doPost(e) {
  try {
    var data = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    return json_(book_(data));
  } catch (err) {
    return json_({ ok: false, error: String(err && err.message || err) });
  }
}

/* ===================== Dostępność ===================== */

function availability_(serviceId, fromStr, days) {
  var svc = CONFIG.SERVICES[serviceId];
  if (!svc) { throw new Error('Nieznany zabieg.'); }
  days = Math.max(1, Math.min(days || 14, 31));
  var tz = CONFIG.TIMEZONE;
  var today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  var from = /^\d{4}-\d{2}-\d{2}$/.test(fromStr || '') ? fromStr : today;
  var lastAllowed = addDays_(today, CONFIG.MAX_DAYS_AHEAD);

  var rangeStart = at_(from, '00:00');
  var rangeEnd = at_(addDays_(from, days), '00:00');
  var busy = busyIntervals_(rangeStart, rangeEnd);
  var earliest = new Date(Date.now() + CONFIG.MIN_LEAD_HOURS * 3600 * 1000);

  var out = {};
  for (var i = 0; i < days; i++) {
    var d = addDays_(from, i);
    if (d > lastAllowed) { out[d] = []; continue; }
    out[d] = slotsForDay_(d, svc.minutes, busy, earliest);
  }
  return { ok: true, service: serviceId, from: from, days: days, slots: out, maxDate: lastAllowed };
}

function slotsForDay_(dateStr, minutes, busy, earliest) {
  var dow = parseInt(Utilities.formatDate(at_(dateStr, '12:00'), CONFIG.TIMEZONE, 'u'), 10); // 1–7
  var hours = CONFIG.HOURS[dow];
  if (!hours) { return []; }
  var open = at_(dateStr, hours[0]).getTime();
  var close = at_(dateStr, hours[1]).getTime();
  var step = CONFIG.SLOT_STEP_MIN * 60000;
  var dur = minutes * 60000;
  var buf = CONFIG.BUFFER_MIN * 60000;
  var res = [];
  for (var t = open; t + dur <= close; t += step) {
    if (t < earliest.getTime()) { continue; }
    if (isFree_(t, t + dur, busy, buf)) {
      res.push(Utilities.formatDate(new Date(t), CONFIG.TIMEZONE, 'HH:mm'));
    }
  }
  return res;
}

/** Wolne, jeśli wizyta (z przerwą 15 min przed i po) nie nachodzi na żadne wydarzenie. */
function isFree_(start, end, busy, buf) {
  for (var i = 0; i < busy.length; i++) {
    if (start < busy[i][1] + buf && end + buf > busy[i][0]) { return false; }
  }
  return true;
}

function busyIntervals_(start, end) {
  var cal = calendar_();
  var events = cal.getEvents(start, end);
  var list = [];
  events.forEach(function (ev) {
    if (ev.getMyStatus && ev.getMyStatus() === CalendarApp.GuestStatus.NO) { return; } // odrzucone zaproszenia nie blokują
    if (ev.isAllDayEvent()) {
      list.push([ev.getAllDayStartDate().getTime(), ev.getAllDayEndDate().getTime()]);
    } else {
      list.push([ev.getStartTime().getTime(), ev.getEndTime().getTime()]);
    }
  });
  return list;
}

/* ===================== Rezerwacja ===================== */

function book_(d) {
  if (d.website) { return { ok: true }; } // pułapka na boty
  var svc = CONFIG.SERVICES[d.service];
  if (!svc) { throw new Error('Wybierz zabieg.'); }
  var name = clean_(d.name, 80), phone = clean_(d.phone, 30), email = clean_(d.email, 120), notes = clean_(d.notes, 600);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(d.date || '') || !/^\d{2}:\d{2}$/.test(d.time || '')) { throw new Error('Wybierz dzień i godzinę.'); }
  if (name.length < 2) { throw new Error('Podaj imię i nazwisko.'); }
  if (phone.replace(/\D/g, '').length < 9) { throw new Error('Podaj poprawny numer telefonu.'); }
  if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) { throw new Error('Podaj poprawny adres e-mail.'); }
  if (!d.consent) { throw new Error('Zaznacz zgodę na przetwarzanie danych.'); }

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    // ponowne sprawdzenie, czy termin jest nadal wolny
    var free = availability_(d.service, d.date, 1).slots[d.date] || [];
    if (free.indexOf(d.time) === -1) {
      return { ok: false, taken: true, error: 'Ten termin został właśnie zajęty. Wybierz inną godzinę.' };
    }
    var start = at_(d.date, d.time);
    var end = new Date(start.getTime() + svc.minutes * 60000);
    var title = svc.name + ' – ' + name;
    var desc = [
      'Rezerwacja online (strona internetowa)',
      'Zabieg: ' + svc.name + ' (' + svc.minutes + ' min, ' + svc.price + ' zł)',
      'Klientka: ' + name,
      'Telefon: ' + phone,
      'E-mail: ' + email,
      notes ? 'Uwagi: ' + notes : ''
    ].filter(String).join('\n');
    var ev = calendar_().createEvent(title, start, end, { description: desc, location: CONFIG.BUSINESS.address });
    ev.addPopupReminder(60);
    ev.setColor(CalendarApp.EventColor.PALE_RED);
    lock.releaseLock();

    var when = Utilities.formatDate(start, CONFIG.TIMEZONE, 'dd.MM.yyyy') + ', godz. ' + d.time;
    var dayName = dayName_(d.date);
    sendClientMail_(email, name, svc, dayName + ' ' + when, start, end);
    MailApp.sendEmail({
      to: CONFIG.OWNER_EMAIL,
      subject: 'Nowa rezerwacja: ' + svc.name + ' – ' + dayName + ' ' + when,
      body: desc + '\n\nWizyta jest już w Twoim Kalendarzu Google.',
      replyTo: email
    });
    return { ok: true, when: dayName + ' ' + when, service: svc.name };
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function sendClientMail_(email, name, svc, when, start, end) {
  var b = CONFIG.BUSINESS;
  var gcal = 'https://calendar.google.com/calendar/render?action=TEMPLATE' +
    '&text=' + encodeURIComponent(svc.name + ' – ' + b.name) +
    '&dates=' + fmtUtc_(start) + '/' + fmtUtc_(end) +
    '&location=' + encodeURIComponent(b.address) +
    '&details=' + encodeURIComponent('Zmiana lub odwołanie wizyty: tel. ' + b.phone);
  var html =
    '<div style="font-family:Georgia,serif;color:#2E2522;max-width:520px;margin:auto;padding:24px;background:#FBF7F4;border-radius:16px">' +
    '<p style="color:#8A5A4A;letter-spacing:2px;font-size:12px;text-transform:uppercase;margin:0 0 8px">Potwierdzenie rezerwacji</p>' +
    '<h2 style="font-weight:400;margin:0 0 16px">Dziękuję, ' + esc_(name.split(' ')[0]) + '! Do zobaczenia ✦</h2>' +
    '<p style="font-family:Arial,sans-serif;font-size:15px;line-height:1.6;margin:0 0 4px"><b>' + esc_(svc.name) + '</b> · ' + svc.minutes + ' min · ' + svc.price + ' zł</p>' +
    '<p style="font-family:Arial,sans-serif;font-size:15px;line-height:1.6;margin:0 0 16px">' + esc_(when) + '<br>' + esc_(b.address) + '</p>' +
    '<p><a href="' + gcal + '" style="display:inline-block;background:#8A5A4A;color:#fff;text-decoration:none;padding:12px 20px;border-radius:999px;font-family:Arial,sans-serif;font-size:14px">Dodaj do Kalendarza Google</a></p>' +
    '<p style="font-family:Arial,sans-serif;font-size:13px;color:#6B5E59;line-height:1.6">Jeśli musisz zmienić lub odwołać wizytę, zadzwoń lub napisz SMS: <b>' + b.phone + '</b>.<br>Małgorzata Koza – Terapeutka Twarzy</p>' +
    '</div>';
  MailApp.sendEmail({
    to: email,
    subject: 'Potwierdzenie wizyty: ' + svc.name + ' – ' + when,
    htmlBody: html,
    body: 'Potwierdzenie wizyty: ' + svc.name + ', ' + when + ', ' + b.address + '. Zmiana/odwołanie: tel. ' + b.phone,
    name: 'Terapeutka Twarzy · Małgorzata Koza',
    replyTo: CONFIG.OWNER_EMAIL
  });
}

/* ===================== Pomocnicze ===================== */

function calendar_() {
  var c = CONFIG.CALENDAR_ID === 'primary' ? CalendarApp.getDefaultCalendar() : CalendarApp.getCalendarById(CONFIG.CALENDAR_ID);
  if (!c) { throw new Error('Nie znaleziono kalendarza.'); }
  return c;
}
function at_(dateStr, hhmm) { return Utilities.parseDate(dateStr + ' ' + hhmm, CONFIG.TIMEZONE, 'yyyy-MM-dd HH:mm'); }
function addDays_(dateStr, n) {
  var d = Utilities.parseDate(dateStr + ' 12:00', CONFIG.TIMEZONE, 'yyyy-MM-dd HH:mm');
  return Utilities.formatDate(new Date(d.getTime() + n * 86400000), CONFIG.TIMEZONE, 'yyyy-MM-dd');
}
function dayName_(dateStr) {
  var names = ['', 'poniedziałek', 'wtorek', 'środa', 'czwartek', 'piątek', 'sobota', 'niedziela'];
  return names[parseInt(Utilities.formatDate(at_(dateStr, '12:00'), CONFIG.TIMEZONE, 'u'), 10)];
}
function fmtUtc_(d) { return Utilities.formatDate(d, 'UTC', "yyyyMMdd'T'HHmmss'Z'"); }
function clean_(s, max) { return String(s || '').replace(/[<>]/g, '').trim().slice(0, max); }
function esc_(s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
